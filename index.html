-- 1. Route Ranking with RANK() OVER Analytic Function

SELECT
    origin_id,
    destination_id,
    origin_id || ' -> ' || destination_id AS route,
    bookings,
    total_revenue,
    RANK() OVER (ORDER BY total_revenue DESC) AS revenue_rank
FROM route_revenue_mv
ORDER BY revenue_rank ASC;

-- 2. Materialized View Refresh PL/SQL Block

BEGIN

-- 3. Daily Revenue with LAG() and SUM() OVER Cumulative Total

SELECT
    booking_day,
    daily_revenue,
    LAG(daily_revenue) OVER (ORDER BY booking_day) AS prev_day_revenue,
    ROUND(daily_revenue - LAG(daily_revenue) OVER (ORDER BY booking_day), 2) AS daily_change,
    SUM(daily_revenue) OVER (
        ORDER BY booking_day 
        ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS running_revenue
FROM daily_revenue_v
ORDER BY booking_day ASC;

-- 1. Flight Search with AT TIME ZONE Dynamic Expression

SELECT
    f.flight_id,
    f.flight_no,
    f.origin_id,
    f.destination_id,
    a1.city AS origin_city,
    a2.city AS destination_city,
    f.fare,
    f.currency,
    TO_CHAR(f.departure_time AT TIME ZONE :target_tz, 'YYYY-MM-DD HH24:MI:SS') AS local_departure,
    TO_CHAR(f.arrival_time AT TIME ZONE :target_tz, 'YYYY-MM-DD HH24:MI:SS') AS local_arrival
FROM Flights f
JOIN Airports a1 ON f.origin_id = a1.airport_id
JOIN Airports a2 ON f.destination_id = a2.airport_id
WHERE (:orig IS NULL OR f.origin_id = :orig)
  AND (:dest IS NULL OR f.destination_id = :dest)
ORDER BY f.departure_time ASC;

-- 2. Atomic Transactional Insert (Bookings + Payments with JSON Metadata)

-- Step 1: Query next primary key IDs
SELECT NVL(MAX(booking_id), 0) + 1 FROM Bookings;
SELECT NVL(MAX(payment_id), 0) + 1 FROM Payments;

-- Step 2: Insert into Bookings (status CONFIRMED, metadata as JSON)
INSERT INTO Bookings (booking_id, passenger_id, flight_id, booking_date, status, metadata)
VALUES (:next_booking_id, :passenger_id, :flight_id, TRUNC(SYSDATE), 'CONFIRMED', :metadata_json);

-- Step 3: Insert into Payments
INSERT INTO Payments (payment_id, booking_id, amount, currency, payment_method, status, payment_date)
VALUES (:next_payment_id, :next_booking_id, :amount, :currency, :method, 'SUCCESS', TRUNC(SYSDATE));

-- Step 4: COMMIT or ROLLBACK
COMMIT;

-- 1. View Definition: BOOKING_DETAILS_V

CREATE OR REPLACE VIEW booking_details_v AS
SELECT 
    b.booking_id,
    p.name AS passenger,
    f.flight_no,
    f.origin_id,
    f.destination_id,
    f.fare,
    b.status,
    b.booking_date
FROM Bookings b
JOIN Passengers p ON b.passenger_id = p.passenger_id
JOIN Flights f ON b.flight_id = f.flight_id;

-- 2. Screen Query with Status, Route & Date Filters

SELECT
    booking_id, passenger, flight_no, origin_id, destination_id,
    origin_id || ' -> ' || destination_id AS route,
    fare, status, booking_date
FROM booking_details_v
WHERE (:status IS NULL OR status = :status)
  AND (:route IS NULL OR origin_id || ' -> ' || destination_id = :route)
ORDER BY booking_id DESC;

-- 3. Cancellation DML

UPDATE Bookings
SET status = 'CANCELLED'
WHERE booking_id = :booking_id;
COMMIT;

-- 1. Extracting JSON Properties using JSON_VALUE

SELECT
    passenger_id,
    name,
    email,
    phone,
    JSON_VALUE(preferences, '$.meal') AS meal,
    JSON_VALUE(preferences, '$.seat') AS seat,
    JSON_VALUE(preferences, '$.baggage' RETURNING NUMBER) AS baggage
FROM Passengers
ORDER BY passenger_id ASC;

-- 2. Grouping and Aggregating on JSON Property

SELECT
    NVL(JSON_VALUE(preferences, '$.meal'), 'Not Specified') AS meal_preference,
    COUNT(*) AS passenger_count
FROM Passengers
GROUP BY JSON_VALUE(preferences, '$.meal')
ORDER BY passenger_count DESC;

Payment Methods Breakdown

Group functions with HAVING COUNT(*) > 10 filter

Method Transactions Total Revenue Avg Transaction
🔍 View SQL (HAVING Clause) 
SELECT
    payment_method,
    COUNT(*) AS total_transactions,
    ROUND(SUM(CASE WHEN currency = 'USD' THEN amount * 83 ELSE amount END), 2) AS total_revenue_inr,
    ROUND(AVG(CASE WHEN currency = 'USD' THEN amount * 83 ELSE amount END), 2) AS avg_transaction_inr
FROM Payments
WHERE status = 'SUCCESS'
GROUP BY payment_method
HAVING COUNT(*) > 10
ORDER BY total_revenue_inr DESC;

Multi-Currency Breakdown (INR vs USD)

Handling transactions in multiple currencies with conversion rate of 83 INR per USD

Currency Transactions Original Currency Total Total (INR Normalized) Avg Transaction (INR)
🔍 View SQL (Multi-Currency Grouping) 
SELECT
    currency,
    COUNT(*) AS payment_count,
    ROUND(SUM(amount), 2) AS original_amount_sum,
    ROUND(SUM(CASE WHEN currency = 'USD' THEN amount * 83 ELSE amount END), 2) AS inr_equivalent_sum,
    ROUND(AVG(CASE WHEN currency = 'USD' THEN amount * 83 ELSE amount END), 2) AS avg_inr_equivalent
FROM Payments
WHERE status = 'SUCCESS'
GROUP BY currency
ORDER BY currency ASC;

Fare Class Classification via Conditional CASE

Segmented into: Budget (< ₹5,000), Standard (₹5,000 - ₹25,000), and Premium (> ₹25,000)

Fare Tier Confirmed Bookings Total Revenue
🔍 View SQL (CASE Expression Breakdown) 
SELECT
    fare_class,
    COUNT(*) AS booking_count,
    ROUND(SUM(fare_inr), 2) AS total_fare_revenue
FROM (
    SELECT
        b.booking_id,
        f.fare AS fare_inr,
        CASE
            WHEN f.fare < 5000 THEN 'Budget (< ₹5,000)'
            WHEN f.fare <= 25000 THEN 'Standard (₹5,000 - ₹25,000)'
            ELSE 'Premium (> ₹25,000)'
        END AS fare_class
    FROM Bookings b
    JOIN Flights f ON b.flight_id = f.flight_id
    WHERE b.status = 'CONFIRMED'
)
GROUP BY fare_class
ORDER BY total_fare_revenue DESC;
